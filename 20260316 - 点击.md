班游，是做一片铺了很多六边形的棋盘游戏。

突然想到，如何辨别是哪块六边形玩家点击了呢。每块的六边形没带button component。

班游说用raycast。
```cs
private void Update()
{
  if (Input.GetMouseButtonDown(0))
  {
    mStartTouchPos = Input.mousePosition;
    mEndTouchPos = new Vector2(10000000f, 10000000f);
    mBoardTouchEvent?.Invoke(TouchPhase.Began, mStartTouchPos);
    RaycastHit hit;
    Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
    int layerval = LayerMask.GetMask("gezi");
    if (Physics.Raycast(ray, out hit, 100f, layerval))
    {
        mStartTouchedObject = hit.collider.gameObject;
        mTileTouchEvent?.Invoke(TileTouchType.TileTouchType_begin, mStartTouchedObject, hit.point);
    }
    mGameUI.disSkillDetailPanel();  
  }
  else if (Input.GetMouseButtonUp(0))
  {
    Vector2 v = Input.mousePosition;
    if (mEndTouchPos == v)
    {
        return;
    }
    mBoardTouchEvent?.Invoke(TouchPhase.Ended, mEndTouchPos);
    mEndTouchPos = v;
    RaycastHit hit;
    Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
    int layerval = LayerMask.GetMask("gezi");
    if (Physics.Raycast(ray, out hit, 100f, layerval))
    {
        mEndTouchedObject = hit.collider.gameObject;
        mTileTouchEvent?.Invoke(TileTouchType.TileTouchType_end, mEndTouchedObject, hit.point);
        if (mStartTouchedObject == hit.collider.gameObject)
        {
            keySelectByTouch(mStartTouchedObject);
            mTileTouchEvent?.Invoke(TileTouchType.TileTouchType_selected, mStartTouchedObject, hit.point);
        }
    }
    else if(Input.GetMouseButton(0))
    {
        mBoardTouchEvent?.Invoke(TouchPhase.Moved,Input.mousePosition);
    }
  }
}
```
