到使用的时候，是这样的。\
用PEvent来invoke包括在array里的method。
```cs
public static void OnPiecesEvent(PEvent e, DataBundle db)
{
    Instance.delegates.Invoke(e, db);
    if (Instance.pointedDelegate.ContainsKey(e))
    {
        Instance.pointedDelegate[e].Invoke(e, db);
    }
    db.recovery()
    Instance.delegatesAfterPEvent.Invoke();
}
```

```cs
/// <summary>
/// 当前最新的单位事件
/// </summary>
private PEvent mCurPiecesEvent = PEvent.PEvent_none;

private Stack<PEvent> mCurPiecesEvents = new Stack<PEvent>();
```

```cs
private void onPiecesEvent(PEvent e, DataBundle data)
{
    mCurPiecesEvent = e;
    mCurPiecesEvents.Push(e);

    if (curEventCount == 0) //如果当前没有事件正在处理，那么将上一次遗留的参数清除
    {
        foreach (Dictionary<string, object> values in mCurEventParams.Values)
        {
            mCurEventParamsPools.Push(values);
            values.Clear();
        }

       mCurEventParams.Clear();
    }

    Dictionary<string, object> eventParams = null;
    if (mCurEventParams.ContainsKey(e))
    {
        eventParams = mCurEventParams[e];
    }
    else
    {
        if (mCurEventParamsPools.Count > 0)
        {
            eventParams = mCurEventParamsPools.Pop();
        }
        else
        {
            eventParams = new Dictionary<string, object>();
        }

        mCurEventParams.Add(e, eventParams);
    }

    eventParams.Clear();
    data.copyData(eventParams);

    curEventCount++;

    switch (e)
    {
        case PEvent.Something1:
            break;
        case PEvent.Something2:
            break;
    }

    curEventCount--;
    //必须保留最新的那个事件参数
    if (curEventCount != 0)
    {
        eventParams.Clear();
        mCurEventParams.Remove(e);
        mCurEventParamsPools.Push(eventParams);
    }

    mCurPiecesEvents.Pop();
    mCurPiecesEvent = mCurPiecesEvents.Count > 0 ? mCurPiecesEvents.Peek() : PEvent.PEvent_none;
}
```
mCurPiecesEvent（技能跟着单位）走的，所以师父才说就这个足够用吧。\
mCurPiecesEvent跟mCurPiecesEvents你怎么看.jpg\
