```cs
static void exportAndroid()
{
  isBuilding = true;

  //解析参数 - 用GetCommandLineArgs读取bash line
  string apkPath = string.Empty;
  string batPath = string.Empty;
  string[] args = System.Environment.GetCommandLineArgs();
  for (int i = 0; i < args.Length; i++)
  {
    if (args[i] == "-apkPath")
    {
        apkPath = args[i + 1];
    }
    else if (args[i] == "-batPath")
    {
        batPath = args[i + 1];
    }
  }
  //如果已拥有所以先删掉？
  FileUtils.DeleteFile(apkPath);

  //版本号递增
  string version = PlayerSettings.bundleVersion;
  //班码在这里对版本号进行了split & +1

  PlayerSettings.bundleVersion = version;
  int n = PlayerSettings.Android.bundleVersionCode;
  PlayerSettings.Android.bundleVersionCode = (n + 1);

  //导入
  //PlayerSettings.Android.keystoreName
  //PlayerSettings.Android.keystorePass
  //PlayerSettings.Android.keyaliasName
  //PlayerSettings.Android.keyaliasPass
  //PlayerSettings.applicationIdentifier
  //PlayerSettings.productName

  EditorUserBuildSettings.buildAppBundle = true;
  PlayerSettings.Android.useAPKExpansionFiles = false;//这是啥

  PlayerSettings.Android.targetArchitectures = AndroidArchitecture.ARM64;
  string defineSymbol = PlayerSettings.GetScriptingDefineSymbolsForGroup(BuildTargetGroup.Android);
  //这里班码在说如果define sysmbol里是amazon平台，有多余处理
  //根据不同平台，代码中加了region来分开不同的代码，应该是这样的处理吧

  //测试GP分包相关
  setDelivery();

  EditorUserBuildSettings.buildAppBundle = false;
  apkPath = apkPath.Replace(".aab", ".apk");
  UnityEditor.Build.Reporting.BuildReport br = BuildPipeline.BuildPlayer(GetBuildScenes(), apkPath, BuildTarget.Android, BuildOptions.None);

  //这里算打包成功了，然后会打印一个bat file，再run这个bat file用于upload到供user下载的地方
  System.Text.StringBuilder sb = new System.Text.StringBuilder();
  //sb.AppendLine("TODO")
  FileUtils.WriteFileByStr(batPath, "upload.bat", sb.ToString());

}
```

```cs

public static void setDelivery()
{
  var assetPackConfig = new AssetPackConfig();
  assetPackConfig.SplitBaseModuleAssets = true;
  AssetPackConfigSerializer.SaveConfig(assetPackConfig);
}
```
