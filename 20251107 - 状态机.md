状态机是一种可以take T Type + T Data的（我称之为类）字典。
```cs
public class StateMachine<TStateType, TShareData> where TStateType : Enum
{
  //TStateType是用作状态机内不同状态的Type，ShareData则是包为了那个Type内用的专用class（例子在后头）

  private IState _currentState;
  private TStateType _currentStateType;
  public TShareData ShareData;

  //我称之为本体的状态机字典
  protected Dictionary<TStateType, IState> _states = new();

  public StateMachine(TShareData shareData)
  {
      ShareData = shareData;
  }

  public void ChangeState(TStateType t)
  {
    if (_states.TryGetValue(t, out var state))
    {
        if (_currentState != null)
        {
            _currentState.OnExit();
        }
        _currentState = state;
        _currentStateType = t;
        //我觉得状态机好用的地方就是wrap除了onEnter和Execute这两个函数，在T Type中定义出要跑的东西，很干净
        _currentState.OnEnter();
        _currentState.Execute();
        }
  }

  public void AddState(TStateType type, IState state)
  {
      _states.Add(type, state);
  }

  public TStateType CurrentStateType => _currentStateType;
}

// 状态接口
public interface IState
{
    void OnEnter();
    void OnExit();
    void Execute();
}

public abstract class AbstractState<TStateType,TShareData>  : IState where TStateType : Enum
{
  protected readonly StateMachine<TStateType,TShareData> _fsm;
  protected bool _animeFinish = false;
  
  protected AbstractState(StateMachine<TStateType,TShareData> fsm)
  {
      _fsm = fsm;
  }
  
  public abstract void OnEnter();
  public abstract void OnExit();
  public abstract void Execute();
}

```

public partial class HotLoadManager
{
  #region Singleton
  public static HotLoadManager Instance
  {
      get
      {
          if (_instance == null)
          {
              _instance = new HotLoadManager();
          }
          return _instance;
      }
  }

  private static HotLoadManager _instance = null;
  #endregion

  private StateMachine<HotLoadStateType, HotLoadRuntimeData> _fsm;
  public HotLoadManager()
  {
    _fsm = new StateMachine<HotLoadStateType, HotLoadRuntimeData>(new HotLoadRuntimeData());
    _fsm.AddState(HotLoadStateType.InitPackage, new InitPackageState(_fsm));
    //如果有更多需要的State就这么加，前面是StateType，后面是State相关的数据class
    _fsm.AddState(HotLoadStateType.UpdateRemoteManifest, new UpdateRemoteManifestState(_fsm));
    _fsm.AddState(HotLoadStateType.CheckDownload, new CheckDownloadState(_fsm));
    _fsm.AddState(HotLoadStateType.Downloading, new DownloadingState(_fsm));
    _fsm.AddState(HotLoadStateType.LoadFinish, new LoadFinishState(_fsm));
    _fsm.AddState(HotLoadStateType.Fail, new FailState(_fsm));
    _fsm.ChangeState(HotLoadStateType.None);
    RegisterEvent();
  }

  public void Init()
  {
      //shareData自如其名，各个State分享用
      _fsm.ShareData.someField = "Default";
  }

  //其他热更新用的函数
  public void StartLoad(bool reload = false,bool hasAnimation = false){}

}

public partial class HotLoadManager
{
  // 初始化开始事件
  public delegate void OnInitialBeginDelegate(InitPackageBeginEvent evt);

  public event OnInitialBeginDelegate OnInitialBegin;

  private void RegisterEvent(){
    HotLoadEventController.Instance.RegisterEvent<InitPackageBeginEvent>((evt) =>
    {
        OnInitialBegin?.Invoke(evt);
    });
  }
  //这样算是连成一组的处理初始化时的函数们

  //其他的State也是这样做delegate + event + 写进registerEvent中在HotLoadManager定义时被连上。
}

  }

    //在处理完一段事情时，也在这里call 状态机的下一个状态
