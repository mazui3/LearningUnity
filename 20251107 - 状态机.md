状态机是一种可以take T Type + T Data的（我称之为类）字典。\
```cs
public class StateMachine<TStateType, TShareData> where TStateType : Enum
{
  //TStateType是用作状态机内不同状态的Type，ShareData则是包为了那个Type内用的专用class（例子在后头）

  private IState _currentState;
  private TStateType _currentStateType;
  public TShareData ShareData;

  //我称之为本体的状态机字典
  protected Dictionary<TStateType, IState> _states = new();

  public StateMachine(TShareData shareData)
  {
      ShareData = shareData;
  }

  public void ChangeState(TStateType t)
  {
    if (_states.TryGetValue(t, out var state))
    {
        if (_currentState != null)
        {
            _currentState.OnExit();
        }
        _currentState = state;
        _currentStateType = t;
        //我觉得状态机好用的地方就是wrap除了onEnter和Execute这两个函数，在T Type中定义出要跑的东西，很干净
        _currentState.OnEnter();
        _currentState.Execute();
        }
  }

  public void AddState(TStateType type, IState state)
  {
      _states.Add(type, state);
  }

  public TStateType CurrentStateType => _currentStateType;
}

```
