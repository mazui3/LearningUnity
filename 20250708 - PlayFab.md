```cs
public static bool IsPlayServicesAvailable()
{
 const string GoogleApiAvailability_Classname =
      "com.google.android.gms.common.GoogleApiAvailability";
  AndroidJavaClass clazz =
      new AndroidJavaClass(GoogleApiAvailability_Classname);
  if (clazz == null)
  {
      Debug.LogWarning("Google Play services is not available, class not found: " + GoogleApiAvailability_Classname);
      return true;
  }
  AndroidJavaObject obj =
            clazz.CallStatic<AndroidJavaObject>("getInstance");
  if(obj == null)
  {
      Debug.LogWarning("Google Play services is not available, instance not found: " + GoogleApiAvailability_Classname);
      return true;
  }

  var androidJC = new AndroidJavaClass("com.unity3d.player.UnityPlayer");
  if(androidJC == null)
  {
      Debug.LogWarning("Google Play services is not available, UnityPlayer class not found");
      return true;
  }
  var activity = androidJC.GetStatic<AndroidJavaObject>("currentActivity");
  if(activity == null)
  {
      Debug.LogWarning("Google Play services is not available, currentActivity not found");
      return true;
  }
  int value = obj.Call<int>("isGooglePlayServicesAvailable", activity);
  // result codes from https://developers.google.com/android/reference/com/google/android/gms/common/ConnectionResult

  // 0 == success
  // 1 == service_missing
  // 2 == update service required
  // 3 == service disabled
  // 18 == service updating
  // 9 == service invalid
  return value == 0;
}
```

```cs
private static void LoginWithDeviceID(bool createAccount, Action<LoginResult> resultCallback,
            Action<PlayFabError> errorCallback)
{
  LogKit.I("开始执行设备ID登录");
#if UNITY_EDITOR
  var request = new PlayFab.ClientModels.LoginWithCustomIDRequest
  {
      CreateAccount = createAccount,
      CustomId = SystemInfo.deviceUniqueIdentifier,
      InfoRequestParameters = GetLoginParams()
  };
#elif UNITY_IPHONE || UNITY_IOS
  var request = new LoginWithIOSDeviceIDRequest
  {
      CreateAccount = createAccount,
      DeviceModel = SystemInfo.deviceModel,
      DeviceId = SystemInfo.deviceUniqueIdentifier,
      InfoRequestParameters = GetLoginParams()
  };
  LogKit.I($"设备登录，AndroidDevice：{SystemInfo.deviceModel}，AndroidDeviceId：{SystemInfo.deviceUniqueIdentifier}");
  PlayFabClientAPI.LoginWithIOSDeviceID(request, resultCallback, errorCallback);
#elif UNITY_ANDROID
  var request = new LoginWithAndroidDeviceIDRequest
  {
     CreateAccount = createAccount,
     AndroidDevice = SystemInfo.deviceModel,
     AndroidDeviceId = SystemInfo.deviceUniqueIdentifier,
     InfoRequestParameters = GetLoginParams()
  };
  LogKit.I($"设备登录，AndroidDevice：{SystemInfo.deviceModel}，AndroidDeviceId：{SystemInfo.deviceUniqueIdentifier}");
  PlayFabClientAPI.LoginWithAndroidDeviceID(request, resultCallback, errorCallback);
#endif
}
```
