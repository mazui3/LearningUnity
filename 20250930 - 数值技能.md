private static float getSkillValByData(PiecesLogic target, PiecesLogic user, Effect_Attribut attriType,
        valueType type, SkillFramConfig skillData, int skilllevel, PiecesLogic pl2 = null)
{

  //这里手动filter target type了 - 写成一个新的function会更易懂呢!
  if (skillData.m_targetType != null && skillData.m_targetType.Count > 0)
  {
    if (!skillData.m_targetType.Contains(-1))
    {
      bool isTarget = false;
      for (int i = 0; i < skillData.m_targetType.Count; i++)
      {
          if (GameConfigData.isPieceTypeOf(target.getPieceType(), (pieces_type)skillData.m_targetType[i]))
          {
              isTarget = true;
              break;
          }
      }

      if (!isTarget) return 0;
    }
  }

  //这个等会回头来看一下
  int ret = mInstance.IsSkillTriggerConditionAchieved(skillData, user, pl2, skilllevel, true);
  if (ret <= 0 || !mInstance.isSkillEffectTriggerCmpt(skillData, user, effectTrigger.effectTrigger_none, pl2))
  {
      return 0;
  }

  //filter SelectCamp
  if (skillData.m_targetCamp == (int)SelectCamp.SelectCamp_self) //技能拥有者才有效
  {
      if (user != target) return 0;
  }
  else if (skillData.m_targetCamp == (int)SelectCamp.SelectCamp_friend)
  {
      if (user.PieceCampId != target.PieceCampId) return 0;
  }
  else if (skillData.m_targetCamp == (int)SelectCamp.SelectCamp_enemy)
  {
      if (user.PieceCampId == target.PieceCampId) return 0;
  }

  //filter target condition
  //这里它一条一条filter了
  //TargetCondition_range
  //TargetCondition_hurt
  //TargetCondition_Morale

  //最后回到三种skill的分类（……）
  //数值类是回归读取表格，读多少加多少，再加技能升级
  if (skillData.m_effectParam != null)
  {
      valueType vt = valueType.valueType_default;
      if (skillData.m_effectParam.ContainsKey("someValue"))
      {
          vt = (valueType)(int)(float)skillData.m_effectParam["someValue"];
      }
  }
  else if (skillData.m_effectType == (int)EffectType.EffectType_function)
  {
    //有特殊的类别单独处理
    if (attriType == Effect_Attribut.Attribut_addCriticalRate &&
                     skillData.m_effectId == (int)Effect_Function.E_Function_absolutelyCrite)
      value = 1000000000;
  }
  else if (skillData.m_effectType == (int)EffectType.EffectType_triggerAction)
  {
  }

  return value;
}
