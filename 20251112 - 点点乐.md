隔壁项目组的游戏除了胡闹厨房like做事点点乐。\
还有传说中很有名，三选一纠正很垃圾状况的，在广告里看过很多的游玩模式。

偷看这纯点点乐like式游戏。\
panel分为两种处理游戏。\
一个panel是mainPanel，应该是显示游戏。\
还有个panel用来处理出动，作为了interactPanel。

在interactPanel里不同的gameObject take care不同的操作，比如。\
DragArea，ClickArea，PinchArea，LongPressArea。\
每个Area自带一个script来处理互动。

做的是客户端游戏，手机的触屏和编辑器不同，测试的时候挺有意思的……\
双指操作写了个编辑器用，滑轮操作来替换。\
但滑动得从game切到simulator…再滑动。

```cs
//This is for PinchArea
private RectTransform targetRect;
private Touch[] touches = new Touch[2];
//大概都是处理distance来看是否user input?
private float _lastDistance;
private float _distance;

private Action<float> OnPinchAction;
private bool _resetFinger = false;

void Update()
{

  if (Input.touchCount == 2) {
      touches[0] = Input.GetTouch(0);
      touches[1] = Input.GetTouch(1);
  
      // 检测两个触控点是否都在目标UI内
      if (IsTouchInUI(touches[0]) && IsTouchInUI(touches[1]))
      {
          _resetFinger = false;
          HandleZoom();
      }
  }
  else
  {
      _distance = 0;
      _lastDistance = 0;
      if (!_resetFinger)
      {
          UIKit.GetPanel<ASMRInteractPanel>().ShowFinger();
          _resetFinger = true;
      }
  }
}

bool IsTouchInUI(Touch touch)
{
    RectTransformUtility.ScreenPointToLocalPointInRectangle(
        targetRect, touch.position, UIKit.Root.Camera, out var localPoint);
    return targetRect.rect.Contains(localPoint/2f);
}

void HandleZoom()
{
  // 计算初始距离（首次触发时记录）
  if (_lastDistance <= 0) {
      _lastDistance = Vector2.Distance(touches[0].position, touches[1].position);
      OnPinchAction?.Invoke(-1); 
      return;
  }
  
  // 当前距离
  var dis = Vector2.Distance(touches[0].position, touches[1].position);
  var delta = Mathf.Abs(_lastDistance - dis);
  _lastDistance = dis;
  _distance += delta;
  OnPinchAction?.Invoke(_distance);
}
```
